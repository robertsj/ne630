% Source frame 3: Review: Kinetics without Multiplication

\Sec{1}{Review: Kinetics without Multiplication}

By first eliminating space from the diffusion equation
and then assuming $\phi(E, t) = \phi(t)\varphi(E)$ for
some known $\varphi(E)$, we found
\begin{equation*}
\tag{25.1 $\to$ 25.3}
 \frac{1}{\bar{v}}\frac{d\phi}{dt}
 + \bar{\Sigma}_t \phi(t)
  = \bar{\Sigma}_s \phi(t)
  + \cancelto{0}{\frac{1}{\textcolor{wcprimary}{k_{\text{eff}}}} \bar{\nu}\bar{\Sigma}_f}  \phi(t) + \textcolor{wcalerted}{S(t)}\, ,
\end{equation*}
or, with $\phi = n''' \bar{v}$,
\begin{equation*}
\tag{Like FNRP 5.2 and 5.7}
 \frac{dn'''}{dt} +  \left ( \frac{1}{l_{\infty}} \right ) n'''(t)  = S(t) \, ,
\end{equation*}
where the mean neutron speed and lifetime are
\begin{equation*}
 \bar v=\frac{\displaystyle \int_0^\infty \varphi(E)\,dE}{\displaystyle \int_0^\infty \frac{\varphi(E)}{v(E)}\,dE}\;
 \quad \text{and} \quad
 l_{\infty} = \frac{1}{\bar{v} \bar{\Sigma}_a} \, ,
\end{equation*}
and the neutron speed is $v(E) \approx 1.384\cdot 10^{6}\sqrt{E} \, \si{\centi\meter\per\second}$ for $E$ in \si{\electronvolt}.

% Source frame 4: Review: The Neutron Lifetime


% Source frame 5: Adding Back Multiplication

\Sec{2}{Adding Back Multiplication}

With multiplication, we have\footnote{For source-free, {\it steady-state} problems, the multiplication factor $\textcolor{wcprimary}{k_{\text{eff}}}$ allowed us to adjust the fission rate and enforce a steady state solution.  For kinetics problems, we need no such adjustment!}
\begin{equation*}
\tag{26.1}
  \frac{1}{\bar{v}}\frac{d\phi}{dt}
 + \bar{\Sigma}_t \phi(t)
  = \bar{\Sigma}_s \phi(t)  +  \bar{\nu}\bar{\Sigma}_f \phi(t) +  S(t)  \, .
\end{equation*}




Let $\bar{\Sigma}_a = \bar{\Sigma}_t - \bar{\Sigma}_s$ and $\phi(t) = n'''(t)\bar{v}$ in (26.1). Then, multiply by volume so that
\begin{align*}
  \frac{dn}{dt}
 &= \bar{v} (\bar{\nu}\bar{\Sigma}_f - \bar{\Sigma}_a) n(t)
  + S(t) \\
 &= \bar{\Sigma}_a\bar{v} \left ( \frac{\bar{\nu}\bar{\Sigma}_f}{\bar{\Sigma}_a} - 1 \right ) n(t)
  + S(t) \\
 \tag{FNRP 5.12}
 \Aboxed{\frac{dn}{dt} &= \left ( \frac{k_{\infty} - 1}{l_{\infty}} \right ) n(t)
  + S(t)} \, .
\end{align*}
Here and below, $n(t)$ is the total neutron population, and $S(t)$ denotes
the volume-integrated source in neutrons/s (rather than neutrons/cm$^3$/s).

% Source frame 6: Kinetics with Multiplication

\Sec{3}{Kinetics with Multiplication}

For finite systems with leakage, the text explains how to incorporate the non-leakage probability $P_{NL}$, and the resulting FNRP (5.22) is identical FNRP (5.12) without the $\infty$'s on $k$ and $l$:
\begin{equation*}
\tag{FNRP 5.22}
\frac{dn}{dt}  = \left ( \frac{k - 1}{l} \right ) n(t) + S(t)
\end{equation*}




{\bf Example}: Let $S(t) = 0$ and $n(0) = n_0$.  Determine $n(t)$.


\textcolor{wcprimary}{
{\small {\bf Ans}.  The equation is separable, i.e., $dn/n = [(k-1)/l]dt$, and direct integration yields
\begin{equation*}
  \tag{FNRP 5.24}
  \boxed{ n(t)  =  n_0 e^{\left ( \displaystyle \frac{k - 1}{l}  \right ) t} } \, ,
\end{equation*}
or, recalling, $\rho = (k-1)/k$,
\begin{equation*}
  \tag{FNRP 5.24}
  \boxed{ n(t)  =  n_0 e^{ (\rho/\Lambda) t} } \, ,
\end{equation*}
where $\Lambda = l/k$ is the {\bf prompt generation time}.
}}

% Source frame 7: Kinetics of Source-Driven, Subcritical Systems

\Sec{4}{Kinetics of Source-Driven, Subcritical Systems}

When $S(t) \neq 0$, we go from

\begin{equation*}
   \frac{d}{dt} \left ( n(t) e^{-\left(\frac{k-1}{l}\right )t} \right ) = S(t) e^{-\left(\frac{k-1}{l}\right )t} .
\end{equation*}

to

\begin{equation*}
   n(t') =  \frac{\displaystyle\int^{t'}_0  \left( S(t) e^{-\left(\frac{k-1}{l}\right )t} \right ) dt + n(0)}
                 {e^{-\left(\frac{k-1}{l}\right )t'}}
\end{equation*}

If $S(t) = S_0$, then

\begin{equation*}
   n(t) = \frac{l S_0}{k-1}
     \left (  e^{\frac{k-1}{l}t} - 1 \right )
     + n(0) e^{\frac{k-1}{l}t} \, .
\end{equation*}

If we further set $n(0) = 0$, we get FNRP (5.26).

% Source frame 9: Appendix: Solve $dn/dt = [(k-1)/l]n(t)$

\Sec{6}{Appendix: Solving the ODE}

{\bf Problem.} Let $S(t) = 0$ and $n(0) = n_0$. Determine $n(t)$.


  {\small
  {\bf 1. Write Down the Equations}
  \begin{equation*}
   \frac{dn}{dt} =
    \left(\frac{k - 1}{l}\right)n(t),
    \quad n(0) = n_0 \, .
  \end{equation*}



  {\bf 2. Identify Type and Familiar Form}  \\


  This first-order, initial-value problem is {\bf separable}, and we can write
  \begin{equation*}
     \frac{dn}{n(t)} =
     \left(\frac{k - 1}{l}\right) dt \, .
  \end{equation*}
  }



  {\small
  {\bf 3. Integrate Using Appropriate Technique}
  \begin{align*}
    \int^{n(t)}_{n(0)} \frac{dn'}{n'} &=
     \left(\frac{k - 1}{l}\right)
     \int^{t}_{0} dt' \\[4pt]
    \ln\!\left(\frac{n(t)}{n(0)}\right)
      &= \left(\frac{k - 1}{l}\right)t \\[4pt]
    \Rightarrow \quad n(t)
      &= n(0)\, e^{\left(\displaystyle\frac{k - 1}{l}\right)t}
  \end{align*}
  {\bf 4. Apply Initial Conditions}
  \begin{equation*}
    \boxed{ n(t)  =  n_0 e^{\left ( \displaystyle\frac{k - 1}{l}  \right ) t} }
  \end{equation*}
  }

\textbf{With a source:} For constant $k<1$, let $\tau=l/(1-k)$.
The ODE is linear; multiply by the integrating factor $e^{t/\tau}$:
\[
 \frac{dn}{dt}+\frac{n}{\tau}=S(t)
 \quad\Longrightarrow\quad
 \frac{d}{dt}\big(ne^{t/\tau}\big)=S(t)e^{t/\tau}.
\]
Thus, for any specified source history,
\[
 \boxed{n(t)=n(0)e^{-t/\tau}
       +\int_0^t S(t')e^{-(t-t')/\tau}\,dt'.}
\]
For a piecewise source, solve one interval at a time and use the final
population from one interval as the initial condition for the next.

\Sec{7}{An Accelerator-Driven Source}

An accelerator beam strikes a target, producing neutrons that enter a
subcritical assembly. At fixed beam energy, target conditions, and coupling,
assume $S(t)$ is proportional to the beam current. Pulsed targets are widely used
in practice as neutron sources (e.g., \href{https://neutrons-user-facilities.ornl.gov/how-sns-works/}{SNS}), and accelerator-driven, sub-critical reactors
remain an active area of research.

\vspace{5pt}

\textbf{Assumptions:} Fixed spatial shape and spectrum; constant $k$ and $l$;
no feedback or depletion. Treat all fission neutrons as prompt, as in this
lesson. Delayed-neutron precursors and their slow post-pulse tail are omitted.
Use
\[
 k=0.90,\qquad l=10^{-4}\ \mathrm{s},\qquad
 S_0=10^8\ \text{neutrons/s}.
\]
Then $\tau=1$ ms and the steady population for $S=S_0$ is
$n_{\mathrm{ss}}=S_0\tau=10^5$ neutrons. Start each exercise with $n(0)=0$.

\columnbreak

\Sec{8}{Time-Dependent Sources: ODE Practice}

\textbf{Example 1: One beam pulse.} Let $S(t)=S_0$ for $0\leq t<T$,
then turn the beam off, with $T=2$ ms. Determine $n(t)$ during and after
the pulse. Is $n(t)$ continuous at $T$? Is $dn/dt$?

\vspace{5pt}

{\color{wcprimary}\small
\textbf{Ans.} Solve the constant-source problem, then use $n(T)$ as the
initial condition for the source-free decay:
\[
 n(t)=
 \begin{cases}
 S_0\tau(1-e^{-t/\tau}), & 0\leq t\leq T,\\[3pt]
 S_0\tau(1-e^{-T/\tau})e^{-(t-T)/\tau}, & t>T.
 \end{cases}
\]
The population is continuous, but its derivative jumps by $-S_0$ at beam
shutoff. Check: $n(T)=8.65\times10^4$ and
$n(T+3\tau)=4.30\times10^3$ neutrons.\par}


\vfill
\textbf{Example 2: A gradual beam startup.} Ramp the beam current from zero
to its full value over $T=2$ ms:
\[
 S(t)=
 \begin{cases}
 S_0t/T, & 0\leq t\leq T,\\
 S_0, & t>T.
 \end{cases}
\]
Determine $n(t)$ on both intervals. Compare $n(T)$ with Example 1.
Does reaching full beam current imply that the neutron population is
already at steady state?

\vspace{5pt}

{\color{wcprimary}\small
\textbf{Check.} During the ramp,
\[
 n(t)=\frac{S_0\tau}{T}\big[t-\tau(1-e^{-t/\tau})\big].
\]
At $T$, $n(T)=5.68\times10^4$ neutrons. Thereafter, the difference
$n(t)-n_{\mathrm{ss}}$ decays exponentially.\par}

\vfill

\textbf{Example 3: A fading beam.} Let $S(t)=S_0e^{-t/T_s}$.
Obtain $n(t)$ for $T_s\ne\tau$. Then solve the special case $T_s=\tau$
directly, or take the limit of your result. For $T_s=2\tau$, when is
the neutron population largest? Why is the maximum not at $t=0$?

\vspace{5pt}

{\color{wcprimary}\small
\textbf{Check.} When $T_s=\tau$, $n(t)=S_0t e^{-t/\tau}$.
When $T_s=2\tau$, the peak is at $t=2\tau\ln2$ and
$n_{\max}=S_0\tau/2=5.00\times10^4$ neutrons.\par}

\vfill

\textbf{Example 4: A modulated beam.} Let
$S(t)=S_0[1+\epsilon\cos(\omega t)]$, with $0<\epsilon<1$.
Find the long-time periodic response in the form
\[
 n_{\mathrm{per}}(t)=S_0\tau+A\cos(\omega t-\delta).
\]
Determine $A$ and $\delta$, then add the transient needed for $n(0)=0$.
For $\epsilon=0.20$, compare $\omega\tau=0.1$, $1$, and $10$.
Does the neutron population follow the rapid beam modulation?

\vspace{5pt}

{\color{wcprimary}\small
\textbf{Check.} $A/(S_0\tau)=\epsilon/\sqrt{1+(\omega\tau)^2}$ and
$\delta=\tan^{-1}(\omega\tau)$. The three relative amplitudes are
$0.199$, $0.141$, and $0.0199$.\par}
