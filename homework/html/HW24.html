import numpy as np
from ne630_depletion import (
    DAY, YEAR, Cell, UO2Shielding,
    three_group_xs, uo2_initial, solve_depletion,
)

cell = Cell(
    moderator_to_fuel_volume=2.0,             # V_m / V_f
    water_density_g_cm3=0.7,                 # g/cm^3 moderator
    moderator_flux_ratios=(1.0, 1.0, 1.16),  # phi_m / phi_f: fast, intermediate, thermal
    nonleakage=1.0,                          # Apply leakage separately in the LRM.
)
# UO2 density: 10.4 g/cm^3 fuel; fuel diameter: 0.82 cm.
xs3 = three_group_xs(shielding=UO2Shielding(10.4, 0.82))
# Irradiate for 0–3.5 full-power years; YEAR = 365.25*DAY.
times = np.unique(np.r_[0., np.geomspace(60., 30*DAY, 80),
                        np.linspace(30*DAY, 3.5*YEAR, 601)])
# For part (3), fit on 301 evenly spaced burnups from 1 to 40 MWd/kg initial HM.

results = {}
# Gd-157 loadings in atoms/cm^3 fuel, without and with BP.
# 1.0e18 atoms/cm^3 = 1.0e-6 atoms/(barn cm), about 2.61e-4 g/cm^3 of Gd-157.
for label, gd157 in {"without BP": 0.0, "with BP": 1.0e18}.items():
    initial = uo2_initial(
        0.04,                       # U-235 enrichment: 4% by initial uranium mass.
        10.4,                       # UO2 density, g/cm^3 fuel.
        enrichment_basis="weight",
        gd157_atoms_cm3=gd157,
    )
    results[label] = solve_depletion(
        times, initial=initial,
        model=3,                    # Expanded actinide chain.
        include_poisons=True,       # Include explicit I/Xe and Pm/Sm chains.
        power_density_w_cm3=300.0,   # W/cm^3 fuel, held constant.
        spectrum=(0.35, 0.55, 0.10),  # Fixed integrated fractions: fast/intermediate/thermal.
        xs3=xs3, cell=cell,
    )
